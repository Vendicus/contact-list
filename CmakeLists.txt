cmake_minimum_required(VERSION 3.10)
project(KsiazkaAdresowa)

set(CMAKE_CXX_STANDARD 17)

# 1. Wskaż, że pliki wykonywalne mają trafiać do głównego folderu projektu
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR})

# 2. Wskaż pliki źródłowe
add_executable(ksiazka_adresowa
    src/main.cpp 
    src/ksiazka_adresowa.cpp
)

# 3. Wskaż folder z nagłówkami
target_include_directories(ksiazka_adresowa PRIVATE include)
